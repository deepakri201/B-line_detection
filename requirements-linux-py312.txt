#   python3.12 -m venv .venv
#   source .venv/bin/activate
#   pip install -r requirements-linux-py312.txt
#
# The default Linux torch wheel is CUDA 12.1. nvidia-smi must report CUDA 12.1
# or newer. For a driver that only supports CUDA 11.8 through 12.0, install the
# 11.8 wheels first, then this file:
#
#   pip install torch==2.2.2 torchvision==0.17.2 --index-url https://download.pytorch.org/whl/cu118
#
# Torch 2.2 is the first release with Python 3.12 wheels. Torchvision 0.17
# dropped the pretrained= argument, so building a new model in models/ will
# fail. Loading the released segmentation checkpoints with torch.load still works.
# segmentation-models-pytorch stays at 0.2.1 so those checkpoints unpickle.

# utils/labelset_utils.py cannot build training labels on this stack. The Lahey
# scripts use disk and resize, which are unchanged.
